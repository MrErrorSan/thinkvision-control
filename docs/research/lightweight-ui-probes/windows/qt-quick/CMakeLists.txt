cmake_minimum_required(VERSION 3.21)
project(probe_qtquick LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(probe-qtquick WIN32 main.cpp)
qt_add_qml_module(probe-qtquick URI Probe VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(probe-qtquick PRIVATE Qt6::Quick Qt6::QuickControls2)
if(MSVC)
  target_compile_options(probe-qtquick PRIVATE $<$<CONFIG:Release>:/O1 /GL>)
  target_link_options(probe-qtquick PRIVATE $<$<CONFIG:Release>:/LTCG /OPT:REF /OPT:ICF>)
endif()
